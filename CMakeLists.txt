cmake_minimum_required(VERSION 3.24)
project(Aeris VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(AERIS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(AERIS_FUZZ "Build libFuzzer parser target (Clang)" OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
find_package(OpenSSL 3.0 REQUIRED)
find_package(Qt6Keychain 0.14 CONFIG REQUIRED)
find_package(ZXing CONFIG REQUIRED)
if(ZXing_VERSION VERSION_LESS 2.2)
  message(FATAL_ERROR "ZXing-C++ 2.2 or newer is required")
endif()
set(protobuf_MODULE_COMPATIBLE TRUE)
find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
  find_package(Protobuf REQUIRED)
endif()
if(NOT DEFINED VCPKG_TARGET_TRIPLET)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium>=1.0.18)
  pkg_check_modules(ZIP REQUIRED IMPORTED_TARGET libzip>=1.7)
endif()
if(TARGET PkgConfig::SODIUM)
  set(SODIUM_TARGET PkgConfig::SODIUM)
else()
  find_package(unofficial-sodium CONFIG REQUIRED)
  set(SODIUM_TARGET unofficial-sodium::sodium)
endif()
if(TARGET PkgConfig::ZIP)
  set(ZIP_TARGET PkgConfig::ZIP)
else()
  find_package(libzip CONFIG REQUIRED)
  set(ZIP_TARGET libzip::zip)
endif()
protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS src/import/migration.proto)
add_library(aeris_proto STATIC ${PROTO_SRCS} ${PROTO_HDRS})
target_link_libraries(aeris_proto PUBLIC protobuf::libprotobuf-lite)
target_include_directories(aeris_proto PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/import/*.cpp src/storage/*.cpp)
add_library(aeris_core STATIC ${CORE_SOURCES})
target_include_directories(aeris_core PUBLIC src)
target_link_libraries(aeris_core PUBLIC Qt6::Core Qt6::Gui OpenSSL::Crypto ${SODIUM_TARGET} ${ZIP_TARGET} qt6keychain ZXing::ZXing aeris_proto)
target_compile_definitions(aeris_core PUBLIC AERIS_VERSION="${PROJECT_VERSION}")
file(GLOB UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp)
add_library(aeris_ui STATIC ${UI_SOURCES})
target_link_libraries(aeris_ui PUBLIC aeris_core Qt6::Widgets Qt6::Concurrent)
qt_add_executable(aeris WIN32 MACOSX_BUNDLE src/main.cpp assets/resources.qrc assets/brands.qrc)
target_link_libraries(aeris PRIVATE aeris_ui)
set_target_properties(aeris PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.cyberfracture.aeris MACOSX_BUNDLE_BUNDLE_NAME Aeris MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION} MACOSX_BUNDLE_ICON_FILE aeris.icns)
if(APPLE)
  target_sources(aeris PRIVATE assets/aeris.icns)
  set_source_files_properties(assets/aeris.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
if(WIN32)
  configure_file(packaging/windows/aeris.rc.in ${CMAKE_CURRENT_BINARY_DIR}/aeris.rc @ONLY)
  target_sources(aeris PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/aeris.rc)
endif()
foreach(target aeris_core aeris_ui aeris)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(AERIS_SANITIZERS OR AERIS_FUZZ)
  foreach(target aeris_core aeris_ui aeris aeris_proto)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(utility keychain_probe bench_session)
    qt_add_executable(${utility} tests/${utility}.cpp assets/resources.qrc assets/brands.qrc)
    target_link_libraries(${utility} PRIVATE aeris_ui)
  endforeach()
  if(UNIX AND NOT APPLE)
    add_test(NAME keyring_unavailable COMMAND keychain_probe --expect-unavailable)
    set_tests_properties(keyring_unavailable PROPERTIES TIMEOUT 60 ENVIRONMENT "DBUS_SESSION_BUS_ADDRESS=unix:path=/nonexistent-aeris-test-bus")
  endif()
  foreach(suite core import storage ui order)
    qt_add_executable(test_${suite} tests/test_${suite}.cpp)
    target_link_libraries(test_${suite} PRIVATE aeris_ui Qt6::Test)
    target_compile_definitions(test_${suite} PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    if(suite STREQUAL "ui" OR suite STREQUAL "order")
      target_sources(test_${suite} PRIVATE assets/resources.qrc assets/brands.qrc)
    endif()
    add_test(NAME ${suite} COMMAND test_${suite})
    set_tests_properties(${suite} PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
endif()
if(AERIS_FUZZ)
  target_compile_options(aeris_core PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(fuzz_import tests/fuzz_import.cpp)
  target_link_libraries(fuzz_import PRIVATE aeris_core)
  target_compile_options(fuzz_import PRIVATE -fsanitize=fuzzer)
  target_link_options(fuzz_import PRIVATE -fsanitize=fuzzer)
endif()
configure_file(packaging/dependencies.json.in dependencies.json @ONLY)
include(GNUInstallDirs)
install(TARGETS aeris BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/aeris)
install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/aeris/licenses OPTIONAL)
if(UNIX AND NOT APPLE)
  install(FILES packaging/linux/com.cyberfracture.aeris.desktop DESTINATION share/applications)
  install(FILES packaging/linux/com.cyberfracture.aeris.metainfo.xml DESTINATION share/metainfo)
  foreach(size 16 24 32 48 64 128 256 512)
    install(FILES assets/icons/aeris-${size}.png DESTINATION share/icons/hicolor/${size}x${size}/apps RENAME com.cyberfracture.aeris.png)
  endforeach()
endif()
